cmake_minimum_required(VERSION 3.16)

project(Sensors VERSION 0.1 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD_REQUIRED ON)

#find_package(Qt6 6.5 REQUIRED COMPONENTS Quick WebSockets Network)
find_package(Qt6 REQUIRED COMPONENTS Core Quick  WebSockets Widgets WebEngineWidgets WebEngineQuick)

qt_standard_project_setup(REQUIRES 6.5)

qt_add_executable(appSensors
    main.cpp
    socketserver.cpp
    socketserver.h
    responsemanager.cpp
    responsemanager.h
    requestmanager.cpp
    requestmanager.h
    streammanager.cpp
    streammanager.h
    ireqhandler.h
    basereqhandler.cpp
    basereqhandler.h
    core.cpp
    core.h
)

qt_add_qml_module(appSensors
    URI Sensors
    VERSION 1.0
    QML_FILES
        Main.qml
        SOURCES types.h
        SOURCES device_generator.h
)

# Qt for iOS sets MACOSX_BUNDLE_GUI_IDENTIFIER automatically since Qt 6.1.
# If you are developing for iOS or macOS you should consider setting an
# explicit, fixed bundle identifier manually though.
set_target_properties(appSensors PROPERTIES
#    MACOSX_BUNDLE_GUI_IDENTIFIER com.example.appSensors
    MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
    MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION_MAJOR}.${PROJECT_VERSION_MINOR}
    MACOSX_BUNDLE TRUE
    WIN32_EXECUTABLE TRUE
)

target_link_libraries(appSensors
#     PRIVATE Qt6::Quick Qt6::Network Qt6::WebSockets
      PRIVATE Qt6::Quick Qt6::Network Qt6::WebSockets Qt6::Widgets Qt6::WebEngineWidgets Qt6::WebEngineQuick
)

include(GNUInstallDirs)
install(TARGETS appSensors
    BUNDLE DESTINATION .
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
)
